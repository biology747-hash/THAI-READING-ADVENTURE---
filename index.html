/**
 * THAI READING ADVENTURE - Google Apps Script Backend API (Code.gs)
 * โรงเรียนบึงงามพัฒนา | Database: Google Sheets | Storage: Google Drive | AI: Gemini API
 */

const DRIVE_FOLDER_NAME = "THAI_READING_AUDIO_FILES";

function doGet(e) {
  const action = e.parameter.action || "ping";
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  ensureSheetsExist(ss);

  if (action === "getInitialData") {
    return jsonResponse({
      status: "ok",
      students: getSheetRows(ss.getSheetByName("Students")),
      lessons: getSheetRows(ss.getSheetByName("Lessons"))
    });
  }
  return jsonResponse({ status: "ok", message: "THAI READING ADVENTURE API Ready" });
}

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    const action = body.action;
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    ensureSheetsExist(ss);

    if (action === "saveReadingLog") {
      const logSheet = ss.getSheetByName("ReadingLogs");
      let audioDriveUrl = "";
      if (body.audioBase64) {
        audioDriveUrl = saveAudioToDrive(body.studentId, body.lessonId, body.audioBase64);
      }
      logSheet.appendRow([
        new Date().toISOString(),
        body.studentId,
        body.studentName,
        body.lessonId,
        body.targetText,
        body.transcript,
        body.accuracy,
        (body.missedWords || []).join(", "),
        audioDriveUrl
      ]);
      return jsonResponse({ status: "ok", audioDriveUrl: audioDriveUrl });
    }

    if (action === "analyzeReadingAI") {
      // เรียกใช้ Gemini API โดยดึง API Key จาก Script Properties (ไม่เปิดเผยใน Frontend)
      const apiKey = PropertiesService.getScriptProperties().getProperty("GEMINI_API_KEY");
      if (!apiKey) {
        return jsonResponse({ status: "error", message: "กรุณาตั้งค่า GEMINI_API_KEY ใน Script Properties" });
      }
      const prompt = `คุณคือครูภาษาไทยระดับประถมศึกษา วิเคราะห์การอ่านของนักเรียนชั้น ${body.grade}:
ข้อความต้นฉบับ: "${body.targetText}"
ข้อความที่นักเรียนอ่านได้: "${body.transcript}"
คำที่ตกหล่น: "${(body.missedWords || []).join(", ")}"
กรุณาให้คำชม คำแนะนำการออกเสียงที่เข้าใจง่ายสำหรับเด็ก และแบบฝึกหัดสั้นๆ`;

      const url = "https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=" + apiKey;
      const res = UrlFetchApp.fetch(url, {
        method: "post",
        contentType: "application/json",
        payload: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] })
      });
      const data = JSON.parse(res.getContentText());
      const aiText = data.candidates?.[0]?.content?.parts?.[0]?.text || "พยายามได้ดีมากจ้า!";
      return jsonResponse({ status: "ok", aiFeedback: aiText });
    }

    return jsonResponse({ status: "ok" });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.toString() });
  }
}

function saveAudioToDrive(studentId, lessonId, base64Data) {
  const folders = DriveApp.getFoldersByName(DRIVE_FOLDER_NAME);
  const folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(DRIVE_FOLDER_NAME);
  const cleanBase64 = base64Data.split(",").pop();
  const blob = Utilities.newBlob(Utilities.base64Decode(cleanBase64), "audio/webm", `${studentId}_${lessonId}_${Date.now()}.webm`);
  const file = folder.createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return file.getUrl();
}

function ensureSheetsExist(ss) {
  ["Students", "Lessons", "ReadingLogs"].forEach(name => {
    if (!ss.getSheetByName(name)) ss.insertSheet(name);
  });
}

function getSheetRows(sheet) {
  const values = sheet.getDataRange().getValues();
  if (values.length <= 1) return [];
  const headers = values[0];
  return values.slice(1).map(row => {
    let obj = {};
    headers.forEach((h, i) => obj[h] = row[i]);
    return obj;
  });
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

คู่มือการติดตั้งและเผยแพร่ผ่าน GitHub Pages และ Google Apps Script

1. ตั้งค่า Google Sheets & GAS
  1. สร้าง Google Sheets ใหม่ ไปที่เมนู Extensions ➔ Apps Script
  2. วางโค้ด Code.gs ด้านบนแล้วกดบันทึก
  3. ตั้งค่า GEMINI_API_KEY ใน Project Settings ➔ Script Properties
  4. กด Deploy ➔ New deployment ➔ Web app เลือกเป็น Anyone แล้วคัดลอก Web App URL
2. เชื่อมต่อ API Endpoint ในเว็บ
  1. คลิกปุ่มไอคอนก้อนเมฆ มุมขวาบนของเว็บแอป
  2. นำ Web App URL จาก Google Apps Script มาวาง
  3. สามารถเปลี่ยน API Endpoint ได้ตลอดเวลาโดยไม่ต้องแก้โค้ดใหม่
3. เผยแพร่ขึ้น GitHub Pages
  1. สร้าง Repository บน GitHub แล้วอัปโหลดไฟล์ index.html
  2. ไปที่ Settings ➔ Pages เลือก Branch main แล้วกด Save
  3. เปิดใช้งานผ่าน HTTPS เพื่อรองรับไมโครโฟน (Web Speech & MediaRecorder)

ตั้งค่า Backend API (Google Apps Script)

ออกแบบให้สามารถเปลี่ยน API Endpoint ได้ภายหลัง เมื่อใส่ URL ของ Google Apps Script Web App ระบบจะส่งข้อมูลผลการเรียนและไฟล์เสียงขึ้น Google Sheets และ Google Drive อัตโนมัติ
